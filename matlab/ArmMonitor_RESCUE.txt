454:                     end

456:                         m2_id = obj.arm.MOTOR_2_ID;
457:                     end
458:                     try
459:                         obj.arm.disable_motor(m1_id);

468:                 end

471:             end

475:                 if ~isempty(obj.fig) && isgraphics(obj.fig)
